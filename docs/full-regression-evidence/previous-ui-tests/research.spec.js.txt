import {test,expect} from '@playwright/test';
test('page upload, node, fixed branch, compare, move and history',async({page})=>{
 await page.goto('/');
 await page.getByLabel('新的研究目标').fill('独立创意探索 '+Date.now());
 await page.getByRole('button',{name:'创建研究',exact:true}).click();
 await page.locator('.navigation').first().getByText('分支导航与研究节点').click();
 await page.locator('.navigation').last().getByText('研究节点高级操作').click();
 await expect(page.getByRole('heading',{name:'资料与固定输入'})).toBeVisible();
 await page.getByLabel('上传资料',{exact:true}).setInputFiles({name:'idea.txt',mimeType:'text/plain',buffer:Buffer.from('BASE_DOCUMENT')});
 await expect(page.getByRole('checkbox',{name:/idea.txt ready/})).toBeVisible();
 await page.getByRole('checkbox',{name:/idea.txt ready/}).check();
 await page.getByLabel('节点问题 / 续研或分支指令').fill('头脑风暴根节点');
 await page.getByRole('button',{name:'创建节点',exact:true}).click();
 await page.getByRole('button',{name:'执行节点',exact:true}).click();
 await expect(page.locator('.detail .version')).toHaveCount(1);
 await page.locator('.detail .version').click();
 await expect(page.getByRole('heading',{name:'结果 v1'})).toBeVisible();
 await page.locator('.versions').getByRole('checkbox').check();
 await page.getByLabel('节点问题 / 续研或分支指令').fill('A 独立方向');
 await page.getByRole('button',{name:'创建独立分支',exact:true}).click();
 await expect(page.locator('.detail h3').first()).toHaveText('A 独立方向');
 await page.getByRole('button',{name:'执行节点',exact:true}).click();
 await expect(page.locator('.detail .version')).toHaveCount(1);
 const node=page.locator('.react-flow__node').filter({hasText:'A 独立方向'});
 const before=await node.boundingBox();
 await node.hover();await page.mouse.down();await page.mouse.move(before.x+100,before.y+90,{steps:10});await page.mouse.up();
 await expect(page.locator('.detail').getByText('1 份资料 · 1 个结果版本')).toBeVisible();
 await page.getByRole('button',{name:'排除节点',exact:true}).click();
 await expect(page.getByRole('button',{name:'选择节点',exact:true})).toBeVisible();
 await page.getByRole('button',{name:'选择节点',exact:true}).click();
 await page.locator('.versions').filter({hasText:'A 独立方向'}).getByRole('checkbox').check();
 await page.getByLabel('节点问题 / 续研或分支指令').fill('明确比较');
 await page.getByRole('button',{name:'比较所选版本',exact:true}).click();
 await expect(page.locator('.detail h3').first()).toHaveText('明确比较');
 await expect(page.locator('.detail .version')).toHaveCount(1);
 await page.locator('.detail .version').click();
 await expect(page.locator('article').getByText('实际输入 2 个结果 · 1 份资料')).toBeVisible();
 await page.screenshot({path:'../docs/stage-4-ui.png',fullPage:true});
 await page.reload();
 await page.locator('.navigation').first().getByText('分支导航与研究节点').click();
 await expect(page.locator('.react-flow__node')).toHaveCount(4);
});
